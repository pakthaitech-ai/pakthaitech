272: function calcOpportunity(input) {
273:   const quantity = Math.max(1, Math.floor(num(input.quantity) || 1));
274:   const asking = num(input.asking_price);
275:   const targetMargin = Math.max(0, num(input.target_margin_pct) || 20);
276:   const retailPerUnit = num(input.retail_price_pkr);
277: 
278:   const enabled = input.cost_enabled || {};
279:   const travel = enabled.travel === false ? 0 : num(input.travel_cost);
280:   const pickup = enabled.pickup === false ? 0 : num(input.pickup_cost);
281:   const packing = enabled.packing === false ? 0 : num(input.packing_cost);
282:   const freight = enabled.freight === false ? 0 : num(input.freight_cost);
283:   const duties = enabled.duties === false ? 0 : num(input.duties_cost);
284:   const payment = enabled.payment === false ? 0 : num(input.payment_fee);
285:   const pkTransport = enabled.pakistan_transport === false ? 0 : num(input.pakistan_transport_cost);
286:   const contingency = enabled.contingency === false ? 0 : num(input.contingency_cost);
287: 
288:   const purchaseTotal = asking * quantity;
289:   const totalExtra = travel + pickup + packing + freight + duties +
290:     payment + pkTransport + contingency;
291:   const landedTotal = purchaseTotal + totalExtra;
292:   const landedPerUnit = landedTotal / quantity;
293:   const retailTotal = retailPerUnit * quantity;
294: 
295:   const maxLandedTotal = retailTotal * (1 - targetMargin / 100);
296:   const maxPurchaseTotal = Math.max(0, maxLandedTotal - totalExtra);
297:   const maxBuyPerUnit = maxPurchaseTotal / quantity;
298: 
299:   const expectedProfit = asking > 0 ? retailTotal - landedTotal : 0;
300:   const expectedMargin = asking > 0 && retailTotal > 0
301:     ? (expectedProfit / retailTotal) * 100
302:     : 0;
303: 
304:   let classification = asking <= 0 ? "Price Pending" : "Review";
305:   if (asking > 0 && expectedMargin >= targetMargin) classification = "Meets Target Margin";
306:   if (quantity <= 5 && num(input.distance_km) <= 10 && expectedMargin >= targetMargin) {
307:     classification = "Local Small-Lot Opportunity";
308:   }
309:   if (retailTotal <= 0) classification = "Needs Retail Reference";
310: 
311:   return {
312:     quantity,
313:     purchase_total: purchaseTotal,
314:     extra_costs_total: totalExtra,
315:     landed_total: landedTotal,
316:     landed_per_unit: landedPerUnit,
317:     retail_total: retailTotal,
318:     max_purchase_total: maxPurchaseTotal,
319:     max_buy_per_unit: maxBuyPerUnit,
320:     expected_profit: expectedProfit,
321:     expected_margin_pct: expectedMargin,
322:     cost_enabled: enabled,
323:     classification
324:   };
325: }

558:       ON CONFLICT(source_id) DO UPDATE SET

561:         description=excluded.description,
